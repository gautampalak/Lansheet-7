#include <iostream>
#include <cstring>
using namespace std;

int main()
{
    char str[100];
    int i, j = 0;

    cout << "Enter a string: ";
    cin.getline(str, 100);

    for (i = 0; str[i] != '\0'; i++)
    {
        char ch = str[i];

        if (ch != 'a' && ch != 'e' &&
            ch != 'i' && ch != 'o' && ch != 'u' &&
            ch != 'A' && ch != 'E' &&
            ch != 'I' && ch != 'O' && ch != 'U')
        {
            str[j] = str[i];
            j++;
        }
    }

    str[j] = '\0';

    cout << "String after removing vowels = " << str;

    return 0;
}
