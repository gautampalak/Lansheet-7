#include <iostream>
#include <cstring>
using namespace std;

int main()
{
    char str[5][50], temp[50];
    int i, j;

    cout << "Enter 5 strings:\n";

    for (i = 0; i < 5; i++)
        cin.getline(str[i], 50);

    for (i = 0; i < 4; i++)
    {
        for (j = i + 1; j < 5; j++)
        {
            if (strcmp(str[i], str[j]) > 0)
            {
                strcpy(temp, str[i]);
                strcpy(str[i], str[j]);
                strcpy(str[j], temp);
            }
        }
    }

    cout << "Strings in alphabetical order:\n";

    for (i = 0; i < 5; i++)
        cout << str[i] << endl;

    return 0;
}
