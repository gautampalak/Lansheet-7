#include <iostream>
#include <cstring>
using namespace std;

int main()
{
    char str[100];
    int i, vowels = 0;

    cout << "Enter a string: ";
    cin.getline(str, 100);

    for (i = 0; str[i] != '\0'; i++)
    {
        char ch = str[i];

        if (ch == 'a' || ch == 'e' || ch == 'i' ||
            ch == 'o' || ch == 'u' || ch == 'A' ||
            ch == 'E' || ch == 'I' || ch == 'O' ||
            ch == 'U')
        {
            vowels++;
        }
    }

    cout << "Number of vowels = " << vowels;

    return 0;
}
