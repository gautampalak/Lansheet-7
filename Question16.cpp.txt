#include <iostream>
#include <cstring>
using namespace std;

int main()
{
    char str[200];
    int i, words = 0;

    cout << "Enter a string: ";
    cin.getline(str, 200);

    for (i = 0; i < strlen(str); i++)
    {
        if (str[i] != ' ' &&
            (i == 0 || str[i - 1] == ' '))
        {
            words++;
        }
    }

    cout << "Number of words = " << words;

    return 0;
}
