#include <iostream>
#include <cstring>
using namespace std;

int main()
{
    char str1[100], str2[100];
    int i, j, found, len1, len2;

    cout << "Enter first string: ";
    cin.getline(str1, 100);

    cout << "Enter second string: ";
    cin.getline(str2, 100);

    len1 = strlen(str1);
    len2 = strlen(str2);

    if (len1 != len2)
    {
        cout << "Strings are not anagrams.";
        return 0;
    }

    for (i = 0; str1[i] != '\0'; i++)
    {
        found = 0;

        for (j = 0; str2[j] != '\0'; j++)
        {
            if (str1[i] == str2[j])
            {
                str2[j] = '*';
                found = 1;
                break;
            }
        }

        if (found == 0)
            break;
    }

    if (str1[i] == '\0')
        cout << "Strings are anagrams.";
    else
        cout << "Strings are not anagrams.";

    return 0;
}
